#configurando usuário e senha
git config --global user.name [username]
git config --global user.email [git private email: https://github.com/settings/emails]

#adicionando/removendo arquivos ao projeto
git add .

#commitando alterações do projeto
commit -m 'aboba'

#acessa o histórico de commits
git log

#volta para uma versão anterior
git checkout [código hash]

#exibe os branchs do projeto
git branch

#setando o token de acesso para o commit
#https://www.youtube.com/watch?v=IuiH6cBtc58
#git remote set-url origin https://[username]:[token]@github.com/[username]/[repository]
git clone https://[username]:[token]@github.com/[username]/[repository]
git push

#mostra o que foi alterado desde o último commit
git status

#criar novo branch
git checkout -b [nome do branch]

#baixa a versao do branch especificado no computador
git pull origin master
